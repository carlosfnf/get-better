    /* Theme engine (pure UI; app logic lives in the module script below).
       - mode:    auto | light | dark
       - palette: a preset or a custom accent; presets recolor accent AND neutrals.
       Variables are generated from a few numbers (hue/saturation) per mode, so every
       palette is legible in both light and dark. Ember = the stylesheet defaults. */
    (function () {
        var root = document.documentElement;
        var K_MODE = 'nexus-theme', K_PAL = 'nexus-palette', K_CUSTOM = 'nexus-accent';
        function read(k) { try { return localStorage.getItem(k); } catch (e) { return null; } }
        function write(k, v) { try { if (v == null) localStorage.removeItem(k); else localStorage.setItem(k, v); } catch (e) {} }

        var PALETTES = {
            original: { name: 'Original', h: 239, s: 84, nh: 222, ns: 14, explicit: { dark: {
                '--bg-base': '#0B0E14', '--bg-surface': '#151923', '--bg-surface-hover': '#1E2433', '--bg-sunken': '#080A0F',
                '--border': '#2A3143', '--border-strong': '#3B4560',
                '--text-main': '#E2E8F0', '--text-muted': '#94A3B8', '--text-faint': '#64748B',
                '--accent-primary': '#6366F1', '--accent-primary-hover': '#7B7EF5', '--accent-deep': '#3730A3', '--on-accent': '#FFFFFF',
                '--accent-glow': 'rgba(99, 102, 241, .4)', '--accent-soft': 'rgba(99, 102, 241, .12)', '--glow': 'rgba(99, 102, 241, .10)',
                '--accent-secondary': '#06B6D4', '--code-fg': '#60A5FA'
            } } },
            ember:  { name: 'Ember',    h: 18,  s: 82, nh: 30,  ns: 6,  dark: ['#e9783f', '#1b1a18'], light: ['#c24a14', '#fffcf6'] },
            amber:  { name: 'Amber',    h: 42,  s: 88, nh: 40,  ns: 9  },
            forest: { name: 'Forest',   h: 148, s: 52, nh: 150, ns: 9  },
            ocean:  { name: 'Ocean',    h: 207, s: 84, nh: 214, ns: 14 },
            violet: { name: 'Violet',   h: 262, s: 78, nh: 258, ns: 12 },
            rose:   { name: 'Rose',     h: 342, s: 76, nh: 338, ns: 10 },
            mono:   { name: 'Graphite', h: 0,   s: 0,  nh: 0,   ns: 0,  fixedL: { dark: 88, light: 18 } }
        };
        var ORDER = ['original', 'ember', 'amber', 'forest', 'ocean', 'violet', 'rose', 'mono'];
        var VARS = ['--bg-base', '--bg-surface', '--bg-surface-hover', '--bg-sunken', '--border', '--border-strong',
                    '--text-main', '--text-muted', '--text-faint', '--accent-primary', '--accent-primary-hover',
                    '--accent-deep', '--on-accent', '--accent-glow', '--accent-soft', '--accent-secondary',
                    '--code-fg', '--glow'];

        function hsl(h, s, l, a) { return 'hsl(' + Math.round(h) + ' ' + Math.round(s) + '% ' + Math.round(l) + '%' + (a != null ? ' / ' + a : '') + ')'; }
        function clamp(v, lo, hi) { return Math.max(lo, Math.min(hi, v)); }
        function hexToHsl(hex) {
            var m = /^#?([0-9a-f]{6})$/i.exec(hex || ''); if (!m) return null;
            var n = parseInt(m[1], 16), r = (n >> 16 & 255) / 255, g = (n >> 8 & 255) / 255, b = (n & 255) / 255;
            var mx = Math.max(r, g, b), mn = Math.min(r, g, b), l = (mx + mn) / 2, d = mx - mn, h = 0, sat = 0;
            if (d) {
                sat = d / (1 - Math.abs(2 * l - 1));
                h = mx === r ? ((g - b) / d) % 6 : mx === g ? (b - r) / d + 2 : (r - g) / d + 4;
                h = (h * 60 + 360) % 360;
            }
            return { h: h, s: sat * 100, l: l * 100 };
        }
        function hslToRgb(h, s, l) {
            s /= 100; l /= 100;
            var k = function (n) { return (n + h / 30) % 12; }, a = s * Math.min(l, 1 - l);
            var f = function (n) { return l - a * Math.max(-1, Math.min(k(n) - 3, Math.min(9 - k(n), 1))); };
            return [f(0), f(8), f(4)];
        }
        function luminance(rgb) {
            var c = rgb.map(function (v) { return v <= 0.03928 ? v / 12.92 : Math.pow((v + 0.055) / 1.055, 2.4); });
            return 0.2126 * c[0] + 0.7152 * c[1] + 0.0722 * c[2];
        }
        function contrast(l1, l2) { var a = Math.max(l1, l2), b = Math.min(l1, l2); return (a + 0.05) / (b + 0.05); }

        // walk lightness (dir = +1 / -1) until ok(L) holds; returns the last candidate if none does
        function fit(startL, dir, lo, hi, ok) {
            var L = startL;
            while (!ok(L) && L + dir >= lo && L + dir <= hi) L += dir;
            return L;
        }
        var INK = [0.10, 0.07, 0.05];
        // returns {var: value}, or null for the stylesheet defaults
        function build(p, mode) {
            if (!p || p.id === 'ember') return null;
            if (p.explicit && p.explicit[mode]) return p.explicit[mode];
            var dark = mode === 'dark';
            var h = p.h, s = clamp(p.s, 0, 92), nh = p.nh, ns = p.ns;
            var ls = clamp(ns * 2.4, 0, 28);
            var baseRgb = hslToRgb(nh, dark ? ns : ls, dark ? 7 : 94), baseLum = luminance(baseRgb);
            var sunkRgb = hslToRgb(nh, dark ? ns : ls, dark ? 5 : 88), sunkLum = luminance(sunkRgb);

            // accent: text on it must hit 4.5:1 and it must stand off the page at >= 3.3:1
            var L = p.fixedL ? p.fixedL[mode] : fit(dark ? 62 : 44, dark ? 1 : -1, 24, 84, function (l) {
                var a = luminance(hslToRgb(h, s, l));
                return Math.max(contrast(a, 1), contrast(a, luminance(INK))) >= 4.5 && contrast(a, baseLum) >= 3.3;
            });
            var accLum = luminance(hslToRgb(h, s, L));
            var onAcc = contrast(accLum, 1) >= contrast(accLum, luminance(INK)) ? '#ffffff' : '#1a120c';
            // code colour: readable on the recessed surface
            var cs = p.s ? (dark ? 55 : 62) : 0;
            var codeL = fit(dark ? 80 : 28, dark ? 1 : -1, 10, 94, function (l) { return contrast(luminance(hslToRgb(h, cs, l)), sunkLum) >= 5; });

            var v = {};
            if (dark) {
                v['--bg-base'] = hsl(nh, ns, 7);            v['--bg-surface'] = hsl(nh, ns, 10);
                v['--bg-surface-hover'] = hsl(nh, ns, 14);  v['--bg-sunken'] = hsl(nh, ns, 5);
                v['--border'] = hsl(nh, ns, 17);            v['--border-strong'] = hsl(nh, ns, 25);
                v['--text-main'] = hsl(nh, Math.min(ns + 14, 24), 90);
                v['--text-muted'] = hsl(nh, ns * 0.9, 65);  v['--text-faint'] = hsl(nh, ns * 0.8, 46);
                v['--accent-primary-hover'] = hsl(h, s, clamp(L + 7, 0, 97));
                v['--accent-deep'] = hsl(h, s, 28);
                v['--accent-glow'] = hsl(h, s, L, 0.32); v['--accent-soft'] = hsl(h, s, L, 0.11); v['--glow'] = hsl(h, s, L, 0.10);
                v['--accent-secondary'] = hsl((h + 165) % 360, p.s ? 40 : 0, 64);
            } else {
                v['--bg-base'] = hsl(nh, ls, 94);           v['--bg-surface'] = hsl(nh, ls ? clamp(ls + 10, 0, 60) : 0, 98.5);
                v['--bg-surface-hover'] = hsl(nh, ls, 90);  v['--bg-sunken'] = hsl(nh, ls, 88);
                v['--border'] = hsl(nh, ls * 0.9, 83);      v['--border-strong'] = hsl(nh, ls * 0.8, 74);
                v['--text-main'] = hsl(nh, ns ? 26 : 0, 12);
                v['--text-muted'] = hsl(nh, ns ? 14 : 0, 34); v['--text-faint'] = hsl(nh, ns ? 10 : 0, 55);
                v['--accent-primary-hover'] = hsl(h, s, clamp(L + (p.fixedL ? 12 : -6), 0, 97));
                v['--accent-deep'] = hsl(h, s, 24);
                v['--accent-glow'] = hsl(h, s, L, 0.28); v['--accent-soft'] = hsl(h, s, L, 0.09); v['--glow'] = hsl(h, s, L, 0.08);
                v['--accent-secondary'] = hsl((h + 165) % 360, p.s ? 45 : 0, 32);
            }
            v['--accent-primary'] = hsl(h, s, L); v['--on-accent'] = onAcc;
            v['--code-fg'] = hsl(h, cs, codeL);
            return v;
        }
        function customPalette(hex) {
            var c = hexToHsl(hex) || { h: 18, s: 80 };
            var gray = c.s < 8;
            return { id: 'custom', name: 'Custom', h: c.h, s: gray ? 0 : clamp(c.s, 45, 90), nh: c.h, ns: gray ? 0 : 11, fixedL: gray ? { dark: 88, light: 18 } : null };
        }

        var state = {
            mode: (function () { var m = read(K_MODE); return m === 'light' || m === 'auto' ? m : 'dark'; })(),
            palette: read(K_PAL) || 'original',
            custom: read(K_CUSTOM) || '#3aa6a0'
        };
        if (state.palette !== 'custom' && !PALETTES[state.palette]) state.palette = 'ember';
        ORDER.concat(['custom']).forEach(function (id) { if (PALETTES[id]) PALETTES[id].id = id; });

        function effective() {
            if (state.mode !== 'auto') return state.mode;
            return window.matchMedia && matchMedia('(prefers-color-scheme: light)').matches ? 'light' : 'dark';
        }
        function currentPalette() { return state.palette === 'custom' ? customPalette(state.custom) : PALETTES[state.palette]; }

        function apply() {
            if (state.mode === 'auto') root.removeAttribute('data-theme'); else root.setAttribute('data-theme', state.mode);
            var vars = build(currentPalette(), effective());
            VARS.forEach(function (k) { root.style.removeProperty(k); });
            if (vars) Object.keys(vars).forEach(function (k) { root.style.setProperty(k, vars[k]); });
            root.setAttribute('data-palette', state.palette);
            try { var meta = document.querySelector('meta[name=theme-color]'); if (meta) meta.content = getComputedStyle(root).getPropertyValue('--bg-base').trim(); } catch (e) {}
        }
        apply();   // before first paint: no flash
        if (window.matchMedia) {
            var mq = matchMedia('(prefers-color-scheme: light)');
            var onChange = function () { if (state.mode === 'auto') { apply(); ui.paint && ui.paint(); } };
            if (mq.addEventListener) mq.addEventListener('change', onChange); else if (mq.addListener) mq.addListener(onChange);
        }

        var ui = {};
        function swatchColors(id) {
            var mode = effective();
            if (id === 'ember') return PALETTES.ember[mode];
            var p = id === 'custom' ? customPalette(state.custom) : PALETTES[id];
            var v = build(p, mode);
            return [v['--accent-primary'], v['--bg-surface']];
        }
        function setMode(m) { state.mode = m; write(K_MODE, m); apply(); ui.paint(); }
        function setPalette(id) { state.palette = id; write(K_PAL, id); apply(); ui.paint(); }

        document.addEventListener('DOMContentLoaded', function () {
            var themeBtn = document.getElementById('themeBtn');
            var seg = document.getElementById('modeSeg');
            var sw = document.getElementById('swatches');
            var hint = document.getElementById('paletteHint');
            var customIn = document.getElementById('customAccent');
            var customLbl = document.getElementById('customSwatch');

            if (sw) ORDER.forEach(function (id) {
                var b = document.createElement('button');
                b.type = 'button'; b.className = 'swatch'; b.dataset.palette = id;
                b.setAttribute('aria-label', PALETTES[id].name + ' palette'); b.title = PALETTES[id].name;
                b.addEventListener('click', function () { setPalette(id); });
                sw.insertBefore(b, customLbl || null);
            });
            if (customIn) {
                customIn.value = state.custom;
                customIn.addEventListener('input', function () { state.custom = customIn.value; write(K_CUSTOM, state.custom); state.palette = 'custom'; write(K_PAL, 'custom'); apply(); ui.paint(); });
            }
            if (seg) seg.addEventListener('click', function (e) {
                var b = e.target.closest && e.target.closest('button[data-mode]'); if (b) setMode(b.dataset.mode);
            });
            if (themeBtn) themeBtn.addEventListener('click', function () { setMode(effective() === 'light' ? 'dark' : 'light'); });

            ui.paint = function () {
                if (themeBtn) { var l = effective() === 'light'; themeBtn.textContent = l ? '☾' : '☀'; themeBtn.title = l ? 'Switch to dark' : 'Switch to light'; }
                if (seg) Array.prototype.forEach.call(seg.querySelectorAll('button'), function (b) { b.setAttribute('aria-pressed', String(b.dataset.mode === state.mode)); });
                if (sw) Array.prototype.forEach.call(sw.querySelectorAll('.swatch[data-palette]'), function (b) {
                    var c = swatchColors(b.dataset.palette);
                    b.style.background = 'linear-gradient(135deg, ' + c[0] + ' 0 54%, ' + c[1] + ' 54% 100%)';
                    b.setAttribute('aria-pressed', String(b.dataset.palette === state.palette));
                });
                if (customLbl) {
                    var cc = swatchColors('custom');
                    customLbl.style.setProperty('--custom-accent', cc[0]);
                    customLbl.setAttribute('aria-pressed', String(state.palette === 'custom'));
                }
                if (hint) hint.textContent = (state.palette === 'custom' ? 'Custom' : PALETTES[state.palette].name) + ' · ' + (state.mode === 'auto' ? 'follows your system (' + effective() + ')' : state.mode) + '. Saved in this browser.';
            };
            ui.paint();

            // mobile drawer
            var body = document.body;
            function openMenu() { body.classList.add('sidebar-open'); }
            function closeMenu() { body.classList.remove('sidebar-open'); }
            var m = document.getElementById('menuBtn'); if (m) m.addEventListener('click', openMenu);
            var c = document.getElementById('sidebarClose'); if (c) c.addEventListener('click', closeMenu);
            var sc = document.getElementById('scrim'); if (sc) sc.addEventListener('click', closeMenu);
            ['ggufInput', 'draftInput', 'mmprojInput'].forEach(function (id) {
                var el = document.getElementById(id); if (el) el.addEventListener('change', closeMenu);
            });
            document.addEventListener('keydown', function (e) { if (e.key === 'Escape') closeMenu(); });
        });
    })();

/* ================= MARKDOWN (mini, escape-first) ================= */
function escapeHtml(s) {
  return String(s).replace(/[&<>"']/g, c => ({ '&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;' }[c]));
}
function inlineFmt(s) {
  return s
    .replace(/\*\*([^*]+)\*\*/g, '<strong>$1</strong>')
    .replace(/(^|[\s(])\*(\S(?:[^*\n]*\S)?)\*(?=[\s).,!?;:]|$)/g, '$1<em>$2</em>');
}
function formatMarkdown(raw) {
  let text = escapeHtml(raw);
  const codeBlocks = [];
  text = text.replace(/(^|\n)```[^\n]*\n?([\s\S]*?)(\n```|$)/g, (_, nl, code) => {
    codeBlocks.push(code.replace(/\n+$/, ''));
    return `${nl}\x00${codeBlocks.length - 1}\x00`;
  });
  text = text.replace(/`([^`\n]+)`/g, '<code>$1</code>');

  const renderBlock = i => `<pre><code>${codeBlocks[i]}</code></pre>`;
  const lines = text.split('\n');
  let html = '', para = [], listTag = null;
  const flushPara = () => { if (para.length) { html += `<p>${para.join('<br>')}</p>`; para = []; } };
  const closeList = () => { if (listTag) { html += `</${listTag}>`; listTag = null; } };

  for (const line of lines) {
    const t = line.trim();
    if (!t) { flushPara(); closeList(); continue; }
    const ph = t.match(/^\x00(\d+)\x00$/);
    if (ph) { flushPara(); closeList(); html += renderBlock(+ph[1]); continue; }
    const h = t.match(/^(#{1,4})\s+(.*)$/);
    if (h) { flushPara(); closeList(); const l = h[1].length + 2; html += `<h${l}>${inlineFmt(h[2])}</h${l}>`; continue; }
    const ul = t.match(/^[-*+]\s+(.*)$/);
    if (ul) { flushPara(); if (listTag !== 'ul') { closeList(); html += '<ul>'; listTag = 'ul'; } html += `<li>${inlineFmt(ul[1])}</li>`; continue; }
    const ol = t.match(/^(\d+)[.)]\s+(.*)$/);
    if (ol) { flushPara(); if (listTag !== 'ol') { closeList(); html += '<ol>'; listTag = 'ol'; } html += `<li>${inlineFmt(ol[2])}</li>`; continue; }
    const bq = t.match(/^&gt;\s?(.*)$/);
    if (bq) { flushPara(); closeList(); html += `<blockquote>${inlineFmt(bq[1])}</blockquote>`; continue; }
    closeList();
    para.push(inlineFmt(t));
  }
  flushPara(); closeList();
  return html.replace(/\x00(\d+)\x00/g, (_, i) => renderBlock(+i));
}

/* ================= THINK-TAG PARSER ================= */
// Thinking markers by model family: Qwen/DeepSeek style, and Gemma 4's channel format.
const THINK_PAIRS = [
  [THINK_OPEN, THINK_CLOSE],
  ['<|channel>thought', '<channel|>'],
  ['<|channel>thinking', '<channel|>'],
];
const STRAY_MARKERS = /<\|channel>|<channel\|>/g;
function parseThinking(text) {
  // earliest opening marker wins
  let best = null;
  for (const [open, close] of THINK_PAIRS) {
    const i = text.indexOf(open);
    if (i !== -1 && (!best || i < best.start)) best = { start: i, open, close };
  }
  if (!best) {
    // hide a half-streamed opening marker (e.g. "<|chan") until it completes
    let out = text;
    for (const [open] of THINK_PAIRS) {
      for (let n = Math.min(open.length - 1, text.length); n >= 2; n--) {
        if (text.endsWith(open.slice(0, n))) { out = text.slice(0, text.length - n); break; }
      }
      if (out !== text) break;
    }
    return { think: '', response: out.replace(STRAY_MARKERS, '') };
  }
  const { start, open, close } = best;
  const prefix = start > 0 ? text.slice(0, start) : '';
  const end = text.indexOf(close, start + open.length);
  if (end === -1) return { think: text.slice(start + open.length).trim(), response: prefix.replace(STRAY_MARKERS, '').trim() };
  const tail = text.slice(end + close.length);
  const response = (prefix && tail) ? `${prefix}\n\n${tail}` : (prefix || tail);
  return { think: text.slice(start + open.length, end).trim(), response: response.replace(STRAY_MARKERS, '').trim() };
}

/* ================= GGUF PARSER (splitter + shard verification + arch) ================= */
const NEED_MORE = new Error('need-more-bytes');
NEED_MORE.needMore = true;

function parseGgufFromBuffer(ab, fileSize) {
  const dv = new DataView(ab), u8 = new Uint8Array(ab);
  const dec = new TextDecoder();
  let o = 0;
  const need = n => { if (o + n > ab.byteLength) throw NEED_MORE; };

  if (ab.byteLength < 12) throw NEED_MORE;
  if (u8[0] !== 0x47 || u8[1] !== 0x47 || u8[2] !== 0x55 || u8[3] !== 0x46)
    throw new Error('not a GGUF file (bad magic)');
  o = 4;
  const version = dv.getUint32(o, true); o += 4;
  if (version !== 2 && version !== 3) throw new Error(`unsupported GGUF version ${version}`);
  const isV3 = version >= 3;
  need(16);
  const tensorCount = Number(dv.getBigUint64(o, true)); o += 8;
  const kvCount = Number(dv.getBigUint64(o, true)); o += 8;
  if (tensorCount === 0 || tensorCount > 65536) throw new Error(`implausible tensor count ${tensorCount}`);
  if (kvCount > 4096) throw new Error(`implausible metadata count ${kvCount}`);

  const kvStart = o;
  let alignment = 32;
  let arch = null;
  const splitMeta = { no: null, count: null, tensors: null, tensorsCount: null };

  const readLen = () => {
    need(isV3 ? 8 : 4);
    const v = isV3 ? Number(dv.getBigUint64(o, true)) : dv.getUint32(o, true);
    o += isV3 ? 8 : 4;
    return v;
  };
  const readStrRaw = () => {
    const len = readLen();
    if (len > 65536) throw new Error('metadata string too long (corrupt?)');
    need(len);
    const s = o; o += len;
    return { start: s, len };
  };
  const POD_SIZE = { 0:1, 1:1, 2:2, 3:2, 4:4, 5:4, 6:4, 7:1, 10:8, 11:8, 12:8 };
  const skipValue = type => {
    if (POD_SIZE[type] != null) { need(POD_SIZE[type]); o += POD_SIZE[type]; return; }
    if (type === 8) { readStrRaw(); return; }
    if (type === 9) {
      need(4); const et = dv.getUint32(o, true); o += 4;
      const n = readLen();
      if (et === 8) { for (let i = 0; i < n; i++) readStrRaw(); }
      else if (et === 9) { for (let i = 0; i < n; i++) skipValue(9); }
      else {
        if (POD_SIZE[et] == null) throw new Error(`unknown array element type ${et}`);
        need(n * POD_SIZE[et]); o += n * POD_SIZE[et];
      }
      return;
    }
    throw new Error(`unknown GGUF value type ${type}`);
  };

  for (let i = 0; i < kvCount; i++) {
    const ks = readStrRaw();
    need(4);
    const vt = dv.getUint32(o, true); o += 4;
    const key = dec.decode(u8.subarray(ks.start, ks.start + ks.len));
    if (key === 'general.alignment') {
      if (vt !== 4) throw new Error('general.alignment has unexpected type');
      need(4); alignment = dv.getUint32(o, true); o += 4;
      if (alignment < 1 || alignment > (1 << 24)) throw new Error(`implausible alignment ${alignment}`);
    } else if (key === 'general.architecture') {
      if (vt === 8) { const s = readStrRaw(); arch = dec.decode(u8.subarray(s.start, s.start + s.len)); }
      else skipValue(vt);
    } else if (key === 'split.no') {
      if (vt !== 2) throw new Error('split.no has unexpected type');
      need(2); splitMeta.no = dv.getUint16(o, true); o += 2;
    } else if (key === 'split.count') {
      if (vt !== 2) throw new Error('split.count has unexpected type');
      need(2); splitMeta.count = dv.getUint16(o, true); o += 2;
    } else if (key === 'split.tensors.count') {
      if (vt !== 5 && vt !== 4) throw new Error('split.tensors.count has unexpected type');
      need(4); splitMeta.tensorsCount = dv.getInt32(o, true); o += 4;
    } else if (key === 'split.tensors') {
      if (vt !== 7) throw new Error('split.tensors has unexpected type');
      need(1); splitMeta.tensors = u8[o] !== 0; o += 1;
    } else {
      skipValue(vt);
    }
  }
  const kvEnd = o;

  const tensors = [];
  for (let i = 0; i < tensorCount; i++) {
    const ns = readStrRaw();
    const name = dec.decode(u8.subarray(ns.start, ns.start + ns.len));
    need(4); const nDims = dv.getUint32(o, true); o += 4;
    if (nDims < 1 || nDims > 8) throw new Error(`tensor "${name}" has bad dim count`);
    need(nDims * 8);
    const dims = [];
    for (let d = 0; d < nDims; d++) { dims.push(Number(dv.getBigUint64(o, true))); o += 8; }
    need(12);
    const type = dv.getUint32(o, true); o += 4;
    const off = Number(dv.getBigUint64(o, true)); o += 8;
    tensors.push({ name, nDims, dims, type, offset: off, size: 0 });
  }

  const dataStart = (o + (alignment - (o % alignment)) % alignment);
  const order = tensors.map((_, i) => i).sort((a, b) => tensors[a].offset - tensors[b].offset);
  for (let k = 0; k < order.length; k++) {
    const t = tensors[order[k]];
    const nextOff = (k + 1 < order.length) ? tensors[order[k + 1]].offset : (fileSize - dataStart);
    t.size = nextOff - t.offset;
    if (t.size < 0 || dataStart + t.offset + t.size > fileSize)
      throw new Error(`tensor "${t.name}" extends past end of file (corrupt GGUF)`);
  }
  return { version, isV3, alignment, arch, kvStart, kvEnd, kvCount, splitMeta, tensors, dataStart, prefix: ab };
}

async function parseGgufStructure(file) {
  if (file.size < 16) throw new Error('file too small to be a GGUF');
  let cap = 1 << 20;
  const hardCap = 192 << 20;
  while (cap <= hardCap) {
    let ab;
    try { ab = await file.slice(0, Math.min(cap, file.size)).arrayBuffer(); }
    catch (e) { throw new Error('could not read file: ' + (e?.message || e)); }
    try { return parseGgufFromBuffer(ab, file.size); }
    catch (e) {
      if (e === NEED_MORE || e?.needMore) { cap *= 8; continue; }
      throw e;
    }
  }
  throw new Error('GGUF metadata too large to parse (over 192 MB) — unsupported');
}

function planShards(struct) {
  const groups = []; let cur = [], curBytes = 0;
  for (const t of struct.tensors) {
    if (curBytes && curBytes + t.size > SPLIT_TARGET_BYTES) { groups.push(cur); cur = []; curBytes = 0; }
    cur.push(t); curBytes += t.size;
  }
  if (cur.length) groups.push(cur);
  return groups;
}

function buildSplitKvs(isV3, shardIdx, totalShards, totalTensors) {
  const enc = new TextEncoder();
  const mk = (key, typeId, valBytes) => {
    const kb = enc.encode(key);
    const b = new Uint8Array((isV3 ? 8 : 4) + kb.length + 4 + valBytes.length);
    const dv = new DataView(b.buffer);
    let o = 0;
    if (isV3) { dv.setBigUint64(0, BigInt(kb.length), true); o = 8; }
    else { dv.setUint32(0, kb.length, true); o = 4; }
    b.set(kb, o); o += kb.length;
    dv.setUint32(o, typeId, true); o += 4;
    b.set(valBytes, o);
    return b;
  };
  const u16 = v => { const b = new Uint8Array(2); new DataView(b.buffer).setUint16(0, v, true); return b; };
  const i32 = v => { const b = new Uint8Array(4); new DataView(b.buffer).setInt32(0, v, true); return b; };
  // same three keys llama.cpp's gguf-split writes (split.tensors.count is REQUIRED by the loader)
  return [
    mk('split.no', 2, u16(shardIdx)),
    mk('split.count', 2, u16(totalShards)),
    mk('split.tensors.count', 5, i32(totalTensors)),
  ];
}

function buildShardHeadParts(struct, group, shardIdx, totalShards) {
  const { isV3, alignment } = struct;
  let running = 0;
  for (const t of group) {
    t._local = running;
    t._pad = (alignment - (t.size % alignment)) % alignment;
    running += t.size + t._pad;
  }
  const enc = new TextEncoder();
  let infosLen = 0;
  for (const t of group) infosLen += (isV3 ? 8 : 4) + enc.encode(t.name).length + 4 + 8 * t.nDims + 4 + 8;
  const infos = new Uint8Array(infosLen);
  const idv = new DataView(infos.buffer);
  {
    let o = 0;
    for (const t of group) {
      const nb = enc.encode(t.name);
      if (isV3) { idv.setBigUint64(o, BigInt(nb.length), true); o += 8; }
      else { idv.setUint32(o, nb.length, true); o += 4; }
      infos.set(nb, o); o += nb.length;
      idv.setUint32(o, t.nDims, true); o += 4;
      for (const d of t.dims) { idv.setBigUint64(o, BigInt(d), true); o += 8; }
      idv.setUint32(o, t.type, true); o += 4;
      idv.setBigUint64(o, BigInt(t._local), true); o += 8;
    }
  }
  const head = new Uint8Array(24);
  const hdv = new DataView(head.buffer);
  head[0] = 0x47; head[1] = 0x47; head[2] = 0x55; head[3] = 0x46;
  hdv.setUint32(4, struct.version, true);
  hdv.setBigUint64(8, BigInt(group.length), true);
  hdv.setBigUint64(16, BigInt(struct.kvCount + 3), true);

  const kvRaw = new Uint8Array(struct.prefix, struct.kvStart, struct.kvEnd - struct.kvStart);
  const parts = [head, kvRaw, ...buildSplitKvs(isV3, shardIdx, totalShards, struct.tensors.length), infos];
  const total = parts.reduce((s, p) => s + p.length, 0);
  const pad = (alignment - (total % alignment)) % alignment;
  if (pad) parts.push(new Uint8Array(pad));
  return parts;
}

async function performSplit(file, struct, groups, dirHandlePromise) {
  const baseName = file.name.replace(/\.gguf$/i, '').replace(/-\d{5}-of-\d{5}$/, '');
  const total = groups.length;
  const names = groups.map((_, i) => `${baseName}-${String(i + 1).padStart(5, '0')}-of-${String(total).padStart(5, '0')}.gguf`);
  const zeros = new Uint8Array(struct.alignment);
  const totalBytes = file.size;
  let doneBytes = 0;

  const dir = await dirHandlePromise;
  if (dir === 'cancelled') {
    sysLog('Split cancelled (folder picker closed).', 'warn');
    splitRunDone(names, true);
    return false;
  }

  for (let i = 0; i < groups.length; i++) {
    const group = groups[i];
    const parts = buildShardHeadParts(struct, group, i, total);
    let shardBytes = parts.reduce((s, p) => s + p.length, 0);

    if (dir) {
      const fh = await dir.getFileHandle(names[i], { create: true });
      const w = await fh.createWritable();
      for (const p of parts) await w.write(p);
      for (const t of group) {
        const abs = struct.dataStart + t.offset;
        await w.write(file.slice(abs, abs + t.size));
        if (t._pad) await w.write(zeros.subarray(0, t._pad));
        shardBytes += t.size + t._pad; doneBytes += t.size + t._pad;
        setSplitProgress(i, total, doneBytes, totalBytes);
      }
      await w.close();
      try {
        const wf = await fh.getFile();
        const s2 = await parseGgufStructure(wf);
        if (s2.splitMeta.no !== i || s2.splitMeta.count !== total || s2.tensors.length !== group.length || s2.splitMeta.tensorsCount !== struct.tensors.length)
          throw new Error('split metadata mismatch');
        sysLog(`✂️ ${names[i]} — ${(shardBytes / 1e9).toFixed(2)} GB · ${group.length} tensors · header verified ✓`, 'success');
      } catch (e) {
        sysLog(`✂️ ${names[i]} written, but header re-check failed: ${e?.message || e}`, 'warn');
      }
    } else {
      const blobParts = [...parts];
      for (const t of group) {
        const abs = struct.dataStart + t.offset;
        blobParts.push(file.slice(abs, abs + t.size));
        if (t._pad) blobParts.push(zeros.subarray(0, t._pad));
        shardBytes += t.size + t._pad; doneBytes += t.size + t._pad;
        setSplitProgress(i, total, doneBytes, totalBytes);
      }
      const url = URL.createObjectURL(new Blob(blobParts));
      const a = document.createElement('a');
      a.href = url; a.download = names[i];
      document.body.appendChild(a); a.click(); a.remove();
      setTimeout(() => URL.revokeObjectURL(url), 120000);
      await new Promise(r => setTimeout(r, 1400));
      sysLog(`✂️ ${names[i]} — download started (${(shardBytes / 1e9).toFixed(2)} GB · ${group.length} tensors)`, 'success');
    }
  }
  sysLog(`✂️ split complete: ${total} shards. Select ALL of them together in the loader.`, 'success');
  splitRunDone(names, false);
  return true;
}

async function offerAndSplit(file) {
  let struct;
  try { struct = await parseGgufStructure(file); }
  catch (e) { sysLog(`✂️ cannot split "${file.name}": ${e?.message || e}`, 'error'); return false; }
  if (struct.splitMeta.no != null || struct.splitMeta.count != null || struct.splitMeta.tensors != null || struct.splitMeta.tensorsCount != null) {
    sysLog(`"${file.name}" is already a split shard — select the full set instead of splitting it again.`, 'warn');
    return false;
  }
  const bigTensor = struct.tensors.find(t => t.size > MAX_FILE_BYTES);
  if (bigTensor) {
    sysLog(`✂️ cannot split "${file.name}": tensor "${bigTensor.name}" alone is ${(bigTensor.size / 1e9).toFixed(2)} GB — over the per-shard limit.`, 'error');
    return false;
  }
  const groups = planShards(struct);
  if (groups.length < 2) { sysLog(`"${file.name}" is under the shard target — no split needed.`, 'info'); return false; }

  const answer = await splitPopupAsk(file, struct, groups);
  if (!answer || !answer.go) {
    sysLog(`Split declined for "${file.name}" — it exceeds the single-file limit, so it cannot be loaded as-is.`, 'warn');
    return false;
  }
  await performSplit(file, struct, groups, answer.dirPromise);
  return true;
}

/* ================= SHARD SET VERIFICATION ================= */
async function verifyShardSet(files) {
  if (files.length === 1) {
    try {
      const s = await parseGgufStructure(files[0]);
      if (s.splitMeta.count > 1) {
        sysLog(`"${files[0].name}" is shard ${(s.splitMeta.no ?? 0) + 1} of ${s.splitMeta.count} — select ALL shards together.`, 'error');
        return false;
      }
    } catch {}
    return true;
  }
  const names = new Set();
  let expectedCount = null;
  for (const f of files) {
    let s;
    try { s = await parseGgufStructure(f); }
    catch (e) { sysLog(`shard check: could not parse "${f.name}" (${e?.message || e}) — letting the engine decide.`, 'warn'); return true; }
    if (s.splitMeta.count == null) {
      sysLog(`shard check: "${f.name}" has no split metadata — these don't look like a matching shard set.`, 'warn');
      return true;
    }
    if (expectedCount == null) expectedCount = s.splitMeta.count;
    else if (expectedCount !== s.splitMeta.count) {
      sysLog(`shard check: split.count mismatch in "${f.name}" (${s.splitMeta.count} ≠ ${expectedCount}) — wrong set.`, 'error');
      return false;
    }
    for (const t of s.tensors) {
      if (names.has(t.name)) {
        sysLog(`shard check: tensor "${t.name}" appears in more than one file — invalid set.`, 'error');
        return false;
      }
      names.add(t.name);
    }
  }
  if (expectedCount !== files.length) {
    sysLog(`shard check: expected ${expectedCount} shards, got ${files.length} — select the complete set.`, 'error');
    return false;
  }
  sysLog(`shard check ✓: ${files.length} shards · ${names.size} tensors total · no duplicates.`, 'success');
  return true;
}

/* ================= LEGACY SHARD REPAIR =================
   Shards written by the old splitter lack `split.tensors.count`, which llama.cpp
   refuses to load without. Rebuild just the header in memory and stream the tensor
   data from the original file (lazy Blob.slice — nothing is copied or re-downloaded). */
async function repairLegacyShards(files) {
  if (files.length < 2) return files;
  const structs = [];
  try { for (const f of files) structs.push(await parseGgufStructure(f)); }
  catch (e) { sysLog(`🩹 shard header check failed (${e?.message || e}) — loading files as-is.`, 'warn'); return files; }
  if (!structs.every(s => s.splitMeta.count > 1 && s.splitMeta.tensorsCount == null)) return files;
  const total = structs.reduce((n, s) => n + s.tensors.length, 0);
  sysLog(`🩹 Shards lack split.tensors.count (made by the old splitter) — patching headers in memory (${total} tensors total).`, 'warn');
  const enc = new TextEncoder();
  return files.map((f, idx) => {
    const st = structs[idx];
    const { isV3, alignment } = st;
    const extra = buildSplitKvs(isV3, 0, 1, total)[2];           // only the tensors.count KV
    let infosLen = 0;
    for (const t of st.tensors) infosLen += (isV3 ? 8 : 4) + enc.encode(t.name).length + 4 + 8 * t.nDims + 4 + 8;
    const infos = new Uint8Array(infosLen);
    const idv = new DataView(infos.buffer);
    let o = 0;
    for (const t of st.tensors) {
      const nb = enc.encode(t.name);
      if (isV3) { idv.setBigUint64(o, BigInt(nb.length), true); o += 8; } else { idv.setUint32(o, nb.length, true); o += 4; }
      infos.set(nb, o); o += nb.length;
      idv.setUint32(o, t.nDims, true); o += 4;
      for (const d of t.dims) { idv.setBigUint64(o, BigInt(d), true); o += 8; }
      idv.setUint32(o, t.type, true); o += 4;
      idv.setBigUint64(o, BigInt(t.offset), true); o += 8;
    }
    const head = new Uint8Array(24);
    const hdv = new DataView(head.buffer);
    head.set([0x47, 0x47, 0x55, 0x46], 0);
    hdv.setUint32(4, st.version, true);
    hdv.setBigUint64(8, BigInt(st.tensors.length), true);
    hdv.setBigUint64(16, BigInt(st.kvCount + 1), true);
    const kvRaw = new Uint8Array(st.prefix, st.kvStart, st.kvEnd - st.kvStart);
    const parts = [head, kvRaw, extra, infos];
    const len = parts.reduce((n, p) => n + p.length, 0);
    const pad = (alignment - (len % alignment)) % alignment;
    if (pad) parts.push(new Uint8Array(pad));
    return new Blob([...parts, f.slice(st.dataStart)]);
  });
}

/* ================= URL PRECHECK ================= */
const headCache = new Map();
async function precheckUrl(url, expectWasm) {
  if (headCache.has(url)) return headCache.get(url);
  let reason = null;
  try {
    const r = await fetch(url, { method: 'HEAD' });
    if (!r.ok) reason = `HTTP ${r.status}`;
    else {
      const ct = (r.headers.get('content-type') || '').toLowerCase();
      if (expectWasm && ct && !ct.includes('wasm')) reason = `wrong type (${ct})`;
    }
  } catch { reason = 'fetch error'; }
  headCache.set(url, reason);
  return reason;
}
async function precheckConfig(config) {
  const urls = Object.values(config).filter(v => typeof v === 'string' && /^https?:/.test(v));
  for (const u of urls) {
    const bad = await precheckUrl(u, /\.wasm(\?|#|$)/.test(u));
    if (bad) return `${bad} ← ${u}`;
  }
  return null;
}

async function safeExit(instance, ms = 5000) {
  if (!instance) return;
  try {
    await Promise.race([
      instance.exit(),
      new Promise((_, rej) => setTimeout(() => rej(new Error('exit timeout')), ms)),
    ]);
  } catch {}
}

function loadModelWithWatchdog(instance, files, loadOpts, silentMs) {
  return new Promise((resolve, reject) => {
    let done = false;
    const timer = setInterval(() => {
      if (!done && Date.now() - lastEngineActivity > silentMs) {
        done = true; clearInterval(timer);
        reject(new Error(`stalled (no engine activity for ${Math.round(silentMs / 1000)}s)`));
      }
    }, 5000);
    instance.loadModel(files, loadOpts).then(
      v => { if (!done) { done = true; clearInterval(timer); resolve(v); } },
      e => { if (!done) { done = true; clearInterval(timer); reject(e); } }
    );
  });
}

/* Load params — mode: 'fast' (user perf) | 'safe' (defaults) | 'plain' (v2).
   draft: File → attaches spec_draft_* params (wllama v3 only, verified in
   the 3.6.1 type defs; the blob is appended to loadModel's file list and
   referenced by filename — best-effort wiring with automatic fallback). */
function buildLoadParams(engine, nCtx, isV3, mode, draft) {
  const threads = Math.min(navigator.hardwareConcurrency || 2, 8);
  const p = { n_ctx: nCtx, jinja: true, n_batch: settings.perf.batch };
  if (isV3) {
    p.n_gpu_layers = engine === 'webgpu' ? 99 : 0;
    p.n_threads = engine === 'singlethread' ? 1 : threads;
    p.reasoning_format = 'none';
    if (mode === 'fast') {
      p.flash_attn = settings.perf.flash;
      p.cache_type_k = settings.perf.kv;
      p.cache_type_v = settings.perf.flash ? settings.perf.kv : 'f16';
    }
    if (draft) {
      p.spec_draft_model = draft.name;
      p.spec_draft_n_max = settings.perf.draftMax;
      p.spec_draft_n_min = Math.max(1, Math.floor(settings.perf.draftMax / 3));
      p.spec_draft_p_min = settings.perf.draftPMin;
      p.spec_draft_ngl = engine === 'webgpu' ? 99 : 0;
      p.spec_draft_threads = engine === 'singlethread' ? 1 : Math.min(navigator.hardwareConcurrency || 2, 4);
    }
  } else {
    p.n_threads = engine === 'singlethread' ? 1 : threads;
  }
  return p;
}

async function loadModelFiles(inputFiles) {
  if (isLoading)    { sysLog('A load is already in progress.', 'warn'); return; }
  if (isGenerating) { sysLog('Stop generation before swapping models.', 'warn'); return; }
  const raw = Array.from(inputFiles || []);
  if (!raw.length) return;
  const bad = raw.find(f => !/\.(gguf|bin)$/i.test(f.name));
  if (bad) { sysLog(`"${bad.name}" is not a .gguf file.`, 'warn'); return; }
  // a vision projector selected together with the model files is split off automatically
  const mmPicked = raw.filter(f => /mmproj/i.test(f.name));
  if (mmPicked.length) await setMmproj(mmPicked[0]);
  const mainRaw = raw.filter(f => !/mmproj/i.test(f.name));
  if (!mainRaw.length) {
    sysLog('👁 Only a vision projector was selected — it will be attached on the next model load (♻ Restart, or load a model).', 'info');
    return;
  }
  const files = mainRaw.sort((a, b) => a.name.localeCompare(b.name));

  isLoading = true;
  try {
    const oversized = files.filter(f => f.size > MAX_FILE_BYTES);
    const normal = files.filter(f => f.size <= MAX_FILE_BYTES);
    if (oversized.length) {
      for (const f of oversized) {
        try { await offerAndSplit(f); }
        catch (e) { sysLog(`✂️ split failed for "${f.name}": ${e?.message || e}`, 'error'); }
      }
      if (!normal.length) {
        sysLog('Split flow finished — re-select ALL written shards together in the loader to run the model.', 'info');
        return;
      }
    }
    const loadFiles = normal;
    if (loadFiles.length > 1) {
      if (!(await verifyShardSet(loadFiles))) {
        sysLog('Load aborted: the shard set is invalid or incomplete.', 'error');
        return;
      }
    }
    if (!loadFiles.length) return;

    const nCtx = settings.nCtx;
    const engine = effectiveEngine();
    const displayName = loadFiles.length > 1 ? `${loadFiles[0].name} (+${loadFiles.length - 1} shards)` : loadFiles[0].name;
    const totalMB = (loadFiles.reduce((s, f) => s + f.size, 0) / 1048576).toFixed(0);

    sysLog('--- LOAD SEQUENCE START ---', 'info');
    sysLog(`File: ${displayName} (${totalMB} MB total, ${loadFiles.length} file${loadFiles.length > 1 ? 's' : ''})`);
    sysLog(`Engine: ${ENGINE_LABELS[engine]}${engine === 'webgpu' ? ' (GPU — no COOP/COEP needed)' : ''}`);
    sysLog(`Context: ${nCtx} tokens · preset: ${settings.preset}`);

    // best-effort architecture read (used for the draft-pairing check + info)
    let mainArch = null;
    try { mainArch = (await parseGgufStructure(loadFiles[0])).arch; } catch {}
    if (mainArch) sysLog(`[model] architecture: ${mainArch}`, 'info');
    const engineBase = await repairLegacyShards(loadFiles);

    if (draftFile) {
      if (PREFERRED_VERSION === '2.3.1') {
        sysLog('🎯 Draft model needs wllama v3 — PREFERRED_VERSION is pinned to 2.3.1, draft will be ignored.', 'warn');
      } else {
        if (!draftArch) { try { draftArch = (await parseGgufStructure(draftFile)).arch; } catch {} }
        if (mainArch && draftArch && mainArch !== draftArch) {
          sysLog(`⚠️ Draft architecture "${draftArch}" ≠ main "${mainArch}" — speculative decoding needs a shared tokenizer. This pairing will likely fail or produce garbage.`, 'warn');
        } else if (mainArch && draftArch) {
          sysLog(`🎯 Draft pairing OK: ${draftArch} ⇄ ${draftArch === mainArch ? mainArch : mainArch}`, 'info');
        }
      }
    }

    const dp = desiredPerf();
    sysLog(`⚡ Speed params: FA ${dp.flash ? 'on' : 'off'} · KV ${dp.kvK}/${dp.kvV} · batch ${dp.batch}${draftFile && PREFERRED_VERSION !== '2.3.1' ? ` · draft ${settings.perf.draftMax}-max @ p≥${settings.perf.draftPMin}` : ''}`, 'info');

    // attempt order: (draft first if selected) → v3 fast → v3 safe → v2 legacy
    const attempts = [];
    const useMm = !!mmprojFile && PREFERRED_VERSION !== '2.3.1';
    if (mmprojFile && !useMm) sysLog('👁 Vision projector needs wllama v3 — PREFERRED_VERSION is pinned to 2.3.1, projector will be ignored.', 'warn');
    if (PREFERRED_VERSION !== '2.3.1') {
      if (draftFile) attempts.push({ cand: ENGINE_V3, mode: 'fast', draft: true, mm: useMm });
      attempts.push({ cand: ENGINE_V3, mode: 'fast', draft: false, mm: useMm });
      attempts.push({ cand: ENGINE_V3, mode: 'safe', draft: false, mm: useMm });
      if (useMm) attempts.push({ cand: ENGINE_V3, mode: 'fast', draft: false, mm: false });   // text-only fallback if the projector is rejected
    }
    attempts.push({ cand: ENGINE_V2, mode: 'plain', draft: false });

    let loaded = false, lastErr = null, winner = null, winMode = null, winDraft = false, winMm = false;

    for (let i = 0; i < attempts.length; i++) {
      const a = attempts[i];
      const cand = a.cand;
      const isV3 = cand.version.startsWith('3');
      await safeExit(wllama); wllama = null;

      const modeLabel = [
        modeLabelFor(a.mode, dp),
        a.draft ? '+ 🎯 draft' : '',
        a.mm ? '+ 👁 vision' : '',
      ].filter(Boolean).join(' ');
      sysLog(`[engine] ${i + 1}/${attempts.length} trying wllama v${cand.version} · ${modeLabel}`, 'info');

      const bad = await precheckConfig(cand.config);
      if (bad) { sysLog(`[engine] precheck skip v${cand.version}: ${bad}`, 'warn'); lastErr = new Error(bad); continue; }

      try {
        const mod = await import(cand.entry);
        if (!mod?.Wllama) throw new Error('module exports no Wllama class');
        showProgress(`Initializing engine v${cand.version}…`);

        wllama = new mod.Wllama(cand.config);
        if (isV3 && typeof wllama.isSupportWebGPU === 'function') {
          sysLog(`[engine] v3 WebGPU support check: ${wllama.isSupportWebGPU() ? 'YES' : 'NO'}`);
        }

        const loadOpts = buildLoadParams(engine, nCtx, isV3, a.mode, a.draft ? draftFile : null);
        const engineFiles = [...engineBase, ...(a.draft && draftFile ? [draftFile] : []), ...(a.mm && mmprojFile ? [mmprojFile] : [])];
        if (!isV3) {
          loadOpts.progressCallback = ({ loaded, total }) => {
            const pct = total ? Math.round((loaded / total) * 100) : 100;
            progBar.classList.remove('indeterminate');
            progBar.style.width = `${pct}%`;
            progText.textContent = `Loading weights: ${pct}%`;
          };
        } else {
          showProgress(a.draft ? 'Loading model + draft into memory…' : 'Loading model into memory…', true);
        }

        lastEngineActivity = Date.now();
        engineLoadError = null;
        await loadModelWithWatchdog(wllama, engineFiles, loadOpts, 180000);
        // wllama can resolve even when llama.cpp failed to load the model — verify for real
        if (engineLoadError) throw new Error(engineLoadError);
        if (typeof wllama.isModelLoaded === 'function' && !wllama.isModelLoaded()) throw new Error('engine reported no model loaded');
        loaded = true; winner = cand; winMode = a.mode; winDraft = !!(a.draft && draftFile); winMm = !!(a.mm && mmprojFile);
        break;
      } catch (err) {
        lastErr = err;
        const msg = err?.message || String(err);
        sysLog(`[engine] v${cand.version} (${modeLabel}) failed: ${msg}`, 'warn');
        await safeExit(wllama); wllama = null;
        if (a.draft) sysLog('🎯 Draft wiring rejected — retrying without draft model.', 'warn');
        if (/gguf|tensor|weights|invalid magic/i.test(msg)) {
          sysLog('The model file itself looks invalid/truncated — no engine can load it. Re-download and verify the size.', 'error');
          break;
        }
      }
    }

    if (!loaded) throw lastErr || new Error('engine load failed');
    sysLog(`[engine] WINNER: wllama v${winner.version} · ${winMode}${winDraft ? ' + 🎯 draft' : ''} · ${ENGINE_LABELS[engine]}.`, 'success');
    if (winMm) sysLog(`👁 Vision projector attached (${mmprojFile.name}) — 📎 image input enabled.`, 'success');
    else if (mmprojFile && winner.version.startsWith('3')) sysLog('👁 Vision projector NOT attached (engine rejected it) — running text-only. Paste the [engine] lines above to get this fixed.', 'warn');
    if (winMode === 'safe') {
      sysLog('⚡ Fast params were rejected by this backend — safe defaults used. Your Performance settings will retry on next restart.', 'warn');
    }
    if (winDraft) {
      sysLog('🎯 Speculative decoding ACTIVE — draft model attached. Run ⚡ Speed Test and compare against main-only.', 'success');
    } else if (draftFile && winner.version.startsWith('3')) {
      sysLog('🎯 Draft model NOT attached (engine rejected the wiring) — running main-only. Paste the [engine] lines above to get this fixed.', 'warn');
    }

    if (winner.version.startsWith('3')) {
      try {
        sysLog(`[engine] threads: ${wllama.getNumThreads()} · multi-thread: ${wllama.isMultithread() ? 'yes' : 'no'}`);
      } catch {}
    }

    const isV3winner = winner.version.startsWith('3');
    modelInfo = {
      name: displayName, sizeMB: totalMB, ctx: nCtx, engine, engineVersion: winner.version, arch: mainArch,
      draftName: winDraft ? draftFile.name : null,
      vision: winMm, visionName: winMm ? mmprojFile.name : null,
      draftParams: winDraft ? { max: settings.perf.draftMax, pMin: settings.perf.draftPMin } : null,
      perf: isV3winner
        ? (winMode === 'fast'
            ? { flash: settings.perf.flash, kvK: settings.perf.flash ? settings.perf.kv : 'f16', kvV: settings.perf.flash ? settings.perf.kv : 'f16', batch: settings.perf.batch, overridden: false }
            : { ...desiredPerf(), overridden: true })
        : null,
    };
    loadedFiles = loadFiles;
    modelStatus.textContent = `${displayName} · ${totalMB} MB · ${ENGINE_LABELS[engine]} · ${nCtx} ctx${winDraft ? ' · 🎯' : ''}`;

    finishProgress(true, 'System online');
    history = [];
    const perfTag = isV3winner && winMode === 'fast' ? ` · ⚡ FA ${modelInfo.perf.flash ? 'on' : 'off'} · KV ${modelInfo.perf.kvK}` : '';
    showNote(`Engine online — <strong>${escapeHtml(displayName)}</strong> · ${totalMB} MB · ${ENGINE_LABELS[engine]} · ${nCtx} ctx · wllama v${winner.version}${winDraft ? ` · 🎯 draft ${escapeHtml(draftFile.name)}` : ''}${winMm ? ' · 👁 vision' : ''}${perfTag}`);
    userInput.disabled = false;
    userInput.placeholder = 'Message the model…';
    setSendMode('send');
    benchBtn.disabled = false;
    updateDraftUI();
    updateReloadHint();
    renderTray();
    sysLog('SUCCESS: engine ready. Run ⚡ Speed Test to measure your tok/s.', 'success');
    userInput.focus();
  } catch (err) {
    const msg = err?.message || String(err);
    sysLog(`LOAD ERROR: ${msg}`, 'error');
    finishProgress(false, 'Load failed');
    if (/gguf|invalid magic/i.test(msg)) sysLog('This looks like a bad model file — re-download it and check the size.', 'warn');
    else if (/wasm|stalled/i.test(msg)) sysLog('Engine files unreachable — check the [engine] lines above.', 'warn');
    wllama = null;
    modelStatus.textContent = 'load failed — no model';
    userInput.disabled = true;
    userInput.placeholder = 'Engine offline — load a model first.';
    setSendMode('send');
    benchBtn.disabled = true;
    updateDraftUI();
    updateReloadHint();
  } finally {
    isLoading = false;
  }
}

  if (mode === 'safe') return 'safe defaults';
  return 'legacy';
}

/* ================= 📎 ATTACHMENTS ================= */
const visionReady = () => !!(modelInfo && modelInfo.vision);
const fmtSize = n => n >= 1048576 ? (n / 1048576).toFixed(1) + ' MB' : Math.max(1, Math.round(n / 1024)) + ' KB';

async function prepareImage(file) {
  const url = URL.createObjectURL(file);
  try {
    const img = await new Promise((res, rej) => {
      const i = new Image();
      i.onload = () => res(i);
      i.onerror = () => rej(new Error('this browser could not decode the image'));
      i.src = url;
    });
    const W = img.naturalWidth, H = img.naturalHeight;
    if (!W || !H) throw new Error('image has no size');
    const scale = Math.min(1, MAX_IMG_SIDE / Math.max(W, H));
    const w = Math.max(1, Math.round(W * scale)), h = Math.max(1, Math.round(H * scale));
    const c = document.createElement('canvas');
    c.width = w; c.height = h;
    const ctx = c.getContext('2d');
    ctx.fillStyle = '#fff'; ctx.fillRect(0, 0, w, h);     // flatten transparency
    ctx.drawImage(img, 0, 0, w, h);
    const blob = await new Promise(r => c.toBlob(r, 'image/jpeg', 0.9));
    if (!blob) throw new Error('could not re-encode the image');
    const buf = await blob.arrayBuffer();
    const ts = Math.min(1, 160 / Math.max(w, h));
    const t = document.createElement('canvas');
    t.width = Math.max(1, Math.round(w * ts)); t.height = Math.max(1, Math.round(h * ts));
    t.getContext('2d').drawImage(c, 0, 0, t.width, t.height);
    return { buf, thumb: t.toDataURL('image/jpeg', 0.7), w, h };
  } finally { URL.revokeObjectURL(url); }
}

async function addAttachments(fileList) {
  for (const f of Array.from(fileList || [])) {
    const label = f.name || 'pasted image';
    try {
      if (/\.(gguf|bin)$/i.test(f.name || '')) { sysLog(`📎 "${label}" is a model file — use "Load .GGUF Model" (or 👁 for a projector).`, 'warn'); continue; }
      if ((f.type || '').startsWith('image/')) {
        if (pendingAttachments.filter(a => a.kind === 'image').length >= MAX_IMAGES) { sysLog(`📎 Max ${MAX_IMAGES} images per message.`, 'warn'); continue; }
        const { buf, thumb, w, h } = await prepareImage(f);
        pendingAttachments.push({ kind: 'image', name: f.name || 'pasted-image.png', size: f.size, buf, thumb, w, h });
        sysLog(`📎 Image ready: ${label} (${w}×${h}${w !== 0 && (w === MAX_IMG_SIDE || h === MAX_IMG_SIDE) ? ', downscaled' : ''}).`, 'info');
      } else if ((f.type || '').startsWith('text/') || f.type === 'application/json' || TEXT_EXT.test(f.name || '')) {
        if (pendingAttachments.filter(a => a.kind === 'text').length >= MAX_TEXT_FILES) { sysLog(`📎 Max ${MAX_TEXT_FILES} text files per message.`, 'warn'); continue; }
        if (f.size > MAX_TEXT_FILE_BYTES) { sysLog(`📎 "${label}" is over ${fmtSize(MAX_TEXT_FILE_BYTES)} — too large to paste into the context.`, 'warn'); continue; }
        const text = await f.text();
        if (text.slice(0, 4096).includes('\u0000')) { sysLog(`📎 "${label}" looks binary — only plain-text files can be attached.`, 'warn'); continue; }
        pendingAttachments.push({ kind: 'text', name: f.name, size: f.size, text });
        sysLog(`📎 File ready: ${label} (${fmtSize(f.size)}, ${text.length.toLocaleString()} chars).`, 'info');
      } else {
        sysLog(`📎 "${label}": unsupported file type${/\.(pdf|docx?|xlsx?|pptx?)$/i.test(f.name || '') ? ' — PDF/Office files aren\'t supported yet; export or copy the text into a .txt/.md file' : ''}.`, 'warn');
      }
    } catch (e) {
      sysLog(`📎 Could not attach "${label}": ${e?.message || e}`, 'error');
    }
  }
  renderTray();
}

function renderTray() {
  attachTray.innerHTML = '';
  attachTray.hidden = pendingAttachments.length === 0;
  pendingAttachments.forEach((a, idx) => {
    const chip = document.createElement('div');
    chip.className = 'attach-chip';
    if (a.kind === 'image') { const img = new Image(); img.src = a.thumb; img.alt = ''; chip.appendChild(img); }
    else { const d = document.createElement('div'); d.className = 'doc'; d.textContent = '📄'; chip.appendChild(d); }
    const meta = document.createElement('div'); meta.className = 'meta';
    const nm = document.createElement('div'); nm.className = 'nm'; nm.textContent = a.name; nm.title = a.name;
    const sz = document.createElement('div'); sz.className = 'sz'; sz.textContent = a.kind === 'image' ? `${a.w}×${a.h}` : fmtSize(a.size);
    meta.append(nm, sz); chip.appendChild(meta);
    const x = document.createElement('button'); x.type = 'button'; x.textContent = '✕'; x.title = 'Remove';
    x.addEventListener('click', () => { pendingAttachments.splice(idx, 1); renderTray(); });
    chip.appendChild(x);
    attachTray.appendChild(chip);
  });
  if (pendingAttachments.some(a => a.kind === 'image') && !visionReady()) {
    const w = document.createElement('div'); w.className = 'attach-warn';
    w.textContent = '⚠ Images need a vision projector (👁 in the sidebar, then ♻ Restart) — they can\'t be sent until one is loaded.';
    attachTray.appendChild(w);
  }
  if (!isGenerating) setSendMode('send');
}

function composeUserContent(typed, atts, budgetTokens) {
  const est = s => Math.ceil((s || '').length / 3);
  const imgs = atts.filter(a => a.kind === 'image').length;
  let remaining = Math.max(0, budgetTokens - est(typed) - imgs * IMG_TOKENS - 64) * 3;   // chars
  const blocks = [], truncated = [];
  const textAtts = atts.filter(x => x.kind === 'text');
  for (let idx = 0; idx < textAtts.length; idx++) {
    const a = textAtts[idx];
    const overhead = a.name.length + 40;
    // fair share: unused room from small files rolls over to the later ones
    const room = Math.max(0, Math.floor(remaining / (textAtts.length - idx)) - overhead);
    let body = a.text;
    if (body.length > room) {
      truncated.push(a.name);
      body = body.slice(0, room) + `\n[… truncated: ${(a.text.length - room).toLocaleString()} more characters not shown]`;
    }
    remaining = Math.max(0, remaining - body.length - overhead);
    const fence = body.includes('```') ? '~~~~' : '```';
    blocks.push(`File: ${a.name}\n${fence}\n${body}\n${fence}`);
  }
  const content = blocks.length ? typed + (typed ? '\n\n' : '') + blocks.join('\n\n') : typed;
  return { content, truncated };
}

function toApiMessage(m) {
  if (m.images && m.images.length && visionReady()) {
    // slice(0): the engine may transfer the buffer to its worker; keep our copy intact for later turns
    return { role: m.role, content: [...m.images.map(b => ({ type: 'image', data: b.slice(0) })), { type: 'text', text: m.content }] };
  }
  return { role: m.role, content: m.content };
}

/* ================= CONTEXT MANAGEMENT ================= */
function buildContextMessages() {
  const budget = Math.max(256, (modelInfo?.ctx || 2048) - settings.nPredict - 256);
  const est = s => Math.ceil((s || '').length / 3);
  const out = [];
  let used = 0;
  const sys = systemPromptEl.value.trim();
  if (sys) { out.push({ role: 'system', content: sys }); used += est(sys); }
  const kept = [];
  for (let i = history.length - 1; i >= 0; i--) {
    const m = history[i];
    const t = est(m.content) + (m.images?.length || 0) * IMG_TOKENS;
    if (used + t > budget && kept.length >= 1) break;
    used += t;
    kept.unshift(m);
  }
  return out.concat(kept);
}

/* ================= GENERATION (version-aware adapter) ================= */
async function runGeneration(messages, opts) {
  const isV3 = modelInfo.engineVersion.startsWith('3');

  if (!isV3) {
    let acc = '';
    const result = await wllama.createChatCompletion(messages, {
      nPredict: opts.maxTokens,
      sampling: { temp: opts.temp, top_p: opts.topP, top_k: opts.topK, penalty_repeat: opts.repPenalty, penalty_last_n: 64 },
      onNewToken: (token, piece, currentText, optionals) => {
        acc = currentText;
        opts.onPiece(piece, currentText);
        if (opts.signal?.aborted) { try { optionals?.abortSignal?.(); } catch {} }
      },
    });
    if (typeof result === 'string' && result.trim() && result.length >= acc.length) return result;
    return acc;
  }

  let reasoning = '', content = '';
  const synth = () => (reasoning ? THINK_OPEN + reasoning + THINK_CLOSE : '') + content;

  const req = {
    messages,
    max_tokens: opts.maxTokens,
    temperature: opts.temp,
    top_p: opts.topP,
    top_k: opts.topK,
    penalty_repeat: opts.repPenalty,
    penalty_last_n: 64,
    cache_prompt: true,
    stream: true,
    abortSignal: opts.signal,
  };
  if (!settings.thinking) req.chat_template_kwargs = { enable_thinking: false };

  const result = await wllama.createChatCompletion(req);

  if (result && typeof result[Symbol.asyncIterator] === 'function') {
    for await (const chunk of result) {
      const d = chunk?.choices?.[0]?.delta || {};
      let got = false;
      if (d.reasoning_content) { reasoning += d.reasoning_content; got = true; }
      if (typeof d.content === 'string' && d.content) { content += d.content; got = true; }
      if (got) opts.onPiece(null, synth());
      if (chunk?.usage) opts.onUsage(chunk.usage);
    }
  } else if (result && typeof result === 'object') {
    const msg = result?.choices?.[0]?.message || {};
    if (msg.reasoning_content) reasoning += msg.reasoning_content;
    if (typeof msg.content === 'string') content += msg.content;
    if (result?.usage) opts.onUsage(result.usage);
    if (content || reasoning) opts.onPiece(null, synth());
  }

  if (!content && !reasoning && !opts.signal?.aborted) {
    sysLog('[gen] v3 streaming produced no text — retrying once in non-streaming mode…', 'warn');
    const r2 = await wllama.createChatCompletion({ ...req, stream: false });
    const msg = r2?.choices?.[0]?.message || {};
    if (msg.reasoning_content) reasoning += msg.reasoning_content;
    if (typeof msg.content === 'string') content += msg.content;
    if (r2?.usage) opts.onUsage(r2.usage);
    if (content || reasoning) opts.onPiece(null, synth());
  }

  return synth();
}

async function handleSendMessage() {
  if (isGenerating || isLoading || !wllama) return;
  const typedRaw = userInput.value.trim();
  const atts = pendingAttachments.slice();
  if (!typedRaw && !atts.length) return;
  const imgAtts = atts.filter(a => a.kind === 'image');
  if (imgAtts.length && !visionReady()) {
    sysLog('📎 Can\'t send images: no vision projector is active. Pick one with 👁 in the sidebar (mmproj-*.gguf from your model\'s repo), then ♻ Restart — or remove the images.', 'warn');
    renderTray();
    return;
  }
  const typed = typedRaw || (imgAtts.length > 1 ? 'Describe these images.' : imgAtts.length ? 'Describe this image.' : 'Summarize the attached file.');
  const sysTok = Math.ceil(systemPromptEl.value.trim().length / 3);
  const budget = Math.floor(Math.max(256, (modelInfo?.ctx || 2048) - settings.nPredict - 256 - sysTok) * 0.85);
  const { content, truncated } = composeUserContent(typed, atts, budget);

  userInput.value = '';
  userInput.style.height = 'auto';
  pendingAttachments = [];
  renderTray();
  const entry = {
    role: 'user', content,
    images: imgAtts.map(a => a.buf),
    attachments: atts.map(a => ({ kind: a.kind, name: a.name, size: a.size, thumb: a.thumb })),
  };
  history.push(entry);
  appendUserMessage(typed, entry.attachments);
  if (truncated.length) sysLog(`📎 Truncated to fit the ${modelInfo?.ctx}-token context: ${truncated.join(', ')}. Raise context in Settings (then ♻ Restart) to include more.`, 'warn');

  const { temp, topP, topK, repPenalty, nPredict } = settings;
  const messages = buildContextMessages().map(toApiMessage);
  if (!settings.thinking) {
    for (let i = messages.length - 1; i >= 0; i--) {
      if (messages[i].role === 'user') {
        const c = messages[i].content;
        if (typeof c === 'string') messages[i].content = c + ' /no_think';
        else { const tp = c.find(x => x.type === 'text'); if (tp) tp.text += ' /no_think'; }
        break;
      }
    }
  }

  if (nPredict > (modelInfo.ctx - 256)) sysLog('Warning: max new tokens is close to context size — history trimmed hard.', 'warn');
  sysLog(`Inference start · ${ENGINE_LABELS[modelInfo.engine]}${modelInfo.draftName ? ' + 🎯 draft' : ''} · thinking ${settings.thinking ? 'ON' : 'OFF'} · temp ${temp.toFixed(2)} · top_p ${topP} · top_k ${topK} · nPredict ${nPredict} · ${messages.length} msgs.`, 'info');

  const { bubble } = appendMessage('ai', '');
  isGenerating = true;
  stopRequested = false;
  abortCtrl = new AbortController();
  setSendMode('stop');
  userInput.disabled = true;

  const t0 = performance.now();
  let tokens = 0, current = '', lastPaint = 0, failure = null, finalUsage = null, hintStage = 0;

  const stallChecker = setInterval(() => {
    if (!isGenerating) { clearInterval(stallChecker); return; }
    if (tokens > 0) return;
    const elapsed = (performance.now() - t0) / 1000;
    if (elapsed > 240 && hintStage < 3) {
      hintStage = 3;
      sysLog(`⏳ No tokens after ${Math.round(elapsed)}s — WebGPU looks stuck on this device. Press Esc to stop, then Settings → Advanced → 🐢 Single-Thread CPU → ♻ Restart Engine.`, 'warn');
    } else if (elapsed > 90 && hintStage < 2) {
      hintStage = 2;
      sysLog(`⏳ Still no tokens after ${Math.round(elapsed)}s — first-run WebGPU shader compilation can be very slow on integrated GPUs. Waiting…`, 'warn');
    } else if (elapsed > 30 && hintStage < 1) {
      hintStage = 1;
      sysLog(`⏳ Waiting for first token (${Math.round(elapsed)}s)… Qwen3 thinks before answering; WebGPU also compiles shaders on the first run.`, 'warn');
    }
  }, 5000);

  const scrollIfNear = () => {
    if (chatContainer.scrollHeight - chatContainer.scrollTop - chatContainer.clientHeight < 140)
      chatContainer.scrollTop = chatContainer.scrollHeight;
  };
  const paintLive = () => {
    const prev = bubble.querySelector('details.thought-process');
    const keepOpen = prev ? prev.open : true;
    const { think, response } = parseThinking(current);
    let html = '';
    if (think) html += `<details class="thought-process"${keepOpen ? ' open' : ''}><summary>Thinking…</summary><div class="thought-content">${formatMarkdown(think)}</div></details>`;
    if (response) html += formatMarkdown(response) + '<span class="caret"></span>';
    else if (!think) html += '<p class="pending"><span class="caret"></span></p>';
    const elapsed = (performance.now() - t0) / 1000;
    if (tokens > 2) html += `<div class="response-metrics metrics-live"><span>⏱️ ${elapsed.toFixed(1)}s</span><span>⚡ ${(tokens / Math.max(elapsed, .001)).toFixed(1)} t/s</span></div>`;
    bubble.innerHTML = html;
    scrollIfNear();
  };
  const onPiece = (piece, fullText) => {
    if (tokens === 0) sysLog(`First token after ${((performance.now() - t0) / 1000).toFixed(1)}s.`, 'info');
    tokens++;
    if (typeof fullText === 'string' && fullText.length >= current.length) current = fullText;
    else if (piece) current += piece;
    const now = performance.now();
    if (now - lastPaint > 90) { lastPaint = now; paintLive(); }
  };

  try {
    const finalText = await runGeneration(messages, {
      maxTokens: nPredict, temp, topP, topK, repPenalty,
      signal: abortCtrl.signal,
      onPiece,
      onUsage: (u) => { finalUsage = u; },
    });
    if (typeof finalText === 'string' && finalText.length >= current.length) current = finalText;
  } catch (err) {
    const aborted = stopRequested || abortCtrl.signal.aborted ||
      err?.name === 'AbortError' || /abort/i.test(err?.message || '');
    if (aborted) sysLog('Generation stopped by user.', 'warn');
    else {
      failure = err?.message || String(err);
      sysLog(`INFERENCE ERROR: ${failure}`, 'error');
      if (/kv_cache_full|context/i.test(failure)) sysLog('Context is full — lower max new tokens or hit New Chat.', 'warn');
    }
  }
  clearInterval(stallChecker);

  const duration = (performance.now() - t0) / 1000;
  if (finalUsage?.completion_tokens) tokens = finalUsage.completion_tokens;
  const reasonTok = finalUsage?.completion_tokens_details?.reasoning_tokens;

  const { think, response } = parseThinking(current);
  let final = '';
  if (think) final += `<details class="thought-process"><summary>Thought process · ${duration.toFixed(1)}s</summary><div class="thought-content">${formatMarkdown(think)}</div></details>`;
  if (response) final += formatMarkdown(response);
  if (!think && !response) final += `<p class="pending"><em>${failure ? 'Error: ' + escapeHtml(failure) : 'No output generated.'}</em></p>`;
  else if (failure) final += `<p style="color:var(--danger)"><em>Error: ${escapeHtml(failure)}</em></p>`;
  final += `<div class="response-metrics"><span>⏱️ ${duration.toFixed(1)}s</span><span>⚡ ${duration > 0 ? (tokens / duration).toFixed(1) : '0'} t/s</span><span>🔢 ${tokens} tok</span>${reasonTok ? `<span>🧠 ${reasonTok} think</span>` : ''}${modelInfo.draftName ? '<span>🎯</span>' : ''}</div>`;
  bubble.innerHTML = final;

  if (current.trim()) {
    history.push({ role: 'assistant', content: current });
    const btn = document.createElement('button');
    btn.className = 'copy-btn'; btn.type = 'button'; btn.textContent = 'Copy';
    btn.addEventListener('click', async () => {
      try { await navigator.clipboard.writeText(current); btn.textContent = 'Copied ✓'; }
      catch { btn.textContent = 'Copy failed'; }
      setTimeout(() => (btn.textContent = 'Copy'), 1500);
    });
    bubble.appendChild(btn);
  }

  sysLog(`Done: ${tokens} tokens in ${duration.toFixed(1)}s${reasonTok ? ` (${reasonTok} reasoning)` : ''}${modelInfo.draftName ? ' · 🎯 draft' : ''}.`, 'info');
  isGenerating = false;
  abortCtrl = null;
  userInput.disabled = false;
  setSendMode('send');
  userInput.focus();
  chatContainer.scrollTop = chatContainer.scrollHeight;
}

/* ================= ⚡ SPEED TEST ================= */
const BENCH_PROMPT = 'Describe a storm over the ocean in one vivid paragraph.';

async function runBenchmark() {
  if (!wllama || isGenerating || isLoading) { sysLog('Speed test needs a loaded, idle model.', 'warn'); return; }
  benchBtn.disabled = true;
  sysLog(`⚡ Speed test: generating 64 tokens (thinking off${modelInfo.draftName ? ', draft on' : ''}, chat history untouched)…`, 'info');

  const isV3 = modelInfo.engineVersion.startsWith('3');
  const t0 = performance.now();
  let tokens = 0, timings = null, failure = null;
  try {
    if (isV3) {
      const res = await wllama.createChatCompletion({
        messages: [{ role: 'user', content: BENCH_PROMPT }],
        max_tokens: 64,
        temperature: 0.7,
        top_p: 0.95,
        stream: true,
        cache_prompt: false,
        chat_template_kwargs: { enable_thinking: false },
      });
      if (res && typeof res[Symbol.asyncIterator] === 'function') {
        for await (const ch of res) {
          if (ch?.choices?.[0]?.delta?.content) tokens++;
          if (ch?.timings) timings = ch.timings;
        }
      } else {
        tokens = res?.usage?.completion_tokens ?? 0;
        timings = res?.timings ?? null;
      }
    } else {
      await wllama.createChatCompletion([{ role: 'user', content: BENCH_PROMPT }], {
        nPredict: 64,
        sampling: { temp: 0.7 },
        onNewToken: () => tokens++,
      });
    }
  } catch (e) { failure = e?.message || String(e); }

  const secs = (performance.now() - t0) / 1000;
  const draftTag = modelInfo.draftName ? ' · 🎯 draft' : '';
  if (failure) {
    sysLog(`⚡ Speed test failed: ${failure}`, 'error');
  } else if (timings && (timings.predicted_per_second || timings.prompt_per_second)) {
    sysLog(`⚡ RESULT · prefill: ${timings.prompt_n} tok @ ${(timings.prompt_per_second || 0).toFixed(1)} tok/s · decode: ${timings.predicted_n} tok @ ${(timings.predicted_per_second || 0).toFixed(1)} tok/s${draftTag}`, 'success');
    sysLog(`⚡ (${ENGINE_LABELS[modelInfo.engine]} · FA ${modelInfo.perf?.flash ? 'on' : 'off'} · KV ${modelInfo.perf?.kvK || 'n/a'}${draftTag})`, 'info');
  } else {
    sysLog(`⚡ RESULT: ${tokens} tokens in ${secs.toFixed(1)}s ≈ ${(tokens / Math.max(secs, .001)).toFixed(1)} tok/s overall (includes prompt processing)${draftTag}`, 'success');
  }
  benchBtn.disabled = false;
}
benchBtn.addEventListener('click', runBenchmark);
